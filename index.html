/** * 波波酪梨 · PAYUNi 加解密閘道 * probro-payuni-crypto * * 存在理由：GAS 的 Utilities 只有雜湊，沒有任何對稱加密， * 而 PAYUNi 要求 AES-256-GCM 雙向加密。這一層專責加解密與驗簽， * 不碰任何商業邏輯、不儲存任何資料。 * * 端點： * GET /health 健康檢查 * GET /selftest 用 PAYUNi 官方範例值自我驗證（不需金鑰，寫死測試值） * POST /encrypt 加密建單參數（需 X-Worker-Secret） * POST /decrypt 解密除錯用（需 X-Worker-Secret） * POST /notify 接收 PAYUNi 背景回調，驗簽解密後寫入 Firebase * GET /pay?k= 付款頁（客人從訂單完成頁點進來） * * Secrets（用 npx wrangler secret put 設定）： * PAYUNI_MER_ID 商店代號 * PAYUNI_HASH_KEY Hash Key（32 碼） * PAYUNI_IV_KEY IV Key（16 碼） * WORKER_SHARED_SECRET 自訂密碼，擋住 /encrypt 與 /decrypt * FIREBASE_SERVICE_ACCOUNT service account JSON 全文（單行） * FIREBASE_DB_URL 例：https://probro-stock-default-rtdb.firebaseio.com * GAS_NOTIFY_URL （選填）額外轉發目標，通常留空 */ const enc = new TextEncoder(); out[i] = byte; } return out; } function bytesToB64(u8) { let bin = ''; for (let i = 0; i < u8.length; i++) bin += String.fromCharCode(u8[i]); return btoa(bin); } function b64ToBytes(b64) { const bin = atob(b64); const tag = all.slice(all.length - 16); const joined = `${bytesToB64(cipherText)}:::${bytesToB64(tag)}`; return bytesToHex(enc.encode(joined)); } async function payuniDecrypt(encryptStr, hashKey, ivKey) { const joined = dec.decode(hexToBytes(encryptStr)); merged.set(cipherText); merged.set(tag, cipherText.length); const key = await importAesKey(hashKey, ['decrypt']); const buf = await crypto.subtle.decrypt( { name: 'AES-GCM', iv: enc.encode(ivKey), tagLength: 128 }, key, merged );
